\providecommand{\foldmgridd}[5]{%
  \foreach \gi in {1,...,#4}{\ifnum\gi<#4
    \draw[mgrid] (#1+\gi*#5,#2) -- (#1+\gi*#5,#2+#3*#5);\fi}
  \foreach \gj in {1,...,#3}{\ifnum\gj<#3
    \draw[mgrid] (#1,#2+\gj*#5) -- (#1+#4*#5,#2+\gj*#5);\fi}
  \draw[mframe] (#1,#2) rectangle ++(#4*#5,#3*#5);}
